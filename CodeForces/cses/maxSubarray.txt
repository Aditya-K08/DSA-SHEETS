#include <bits/stdc++.h>
using namespace std;

int main() {
    int n,t;
    cin >> n>>t;
    vector<int> arr(n);

    for (int i = 0; i < n; i++) {
        cin>>arr[i];
    }

    int maxi = 0;
    int curr = 0;
    for(int i=0;i<n;i++){
        curr=max(arr[i],curr+arr[i]);
        maxi=max(curr,maxi);
    }
    cout<<maxi<<endl;
    return 0;
}
