#include<bits/stdc++.h>
using namespace std;

int digitSumFun(int n) {
    int sum = 0;
    while (n > 0) {
        sum += n % 10;
        n /= 10;
    }
    return sum;
}

void solve(){
  int x,t;
  cin>>x>>t;
  int ans=x;
  while (true) {
        if (digitSumFun(ans) % t == 0) {
          cout<<ans;
          break;
        }
        ans++;
    }
  cout<<endl;
}

int main(){
  int test_case;
  cin>>test_case;
  while(test_case-->0){
    solve();
  }

  return 0; 
}
