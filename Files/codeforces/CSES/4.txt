#include <bits/stdc++.h>
using namespace std;


#define f(i,a,n) for(int i=a;i<n;i++)
#define r(i,a,n) for(int i=a;i>=n;i--)
#define ll long long


int main() {
  ll n;
  cin>>n;
  vector<int>arr(n);
  f(i,0,n) cin>>arr[i];
  ll ans=0;
  for (int i = 1; i < n; ++i) {
        if (arr[i] < arr[i - 1]) {
            ans += arr[i - 1] - arr[i];
            arr[i] = arr[i - 1]; 
        }
  }
  cout<<ans;
  return 0;
}
